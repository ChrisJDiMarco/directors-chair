let selectedLight='key',selectedObject='table-1',editTool='translate',editAim=false,editSnap=false;
const currentLight=()=>settings.lights.find(l=>l.id===selectedLight)||settings.lights[0];
function toolsPanel(light=false){return `<h3>Manipulate in 3D</h3><div class="row"><button data-edit="translate" class="${editTool==='translate'&&!editAim?'active':''}">Move · G</button><button data-edit="rotate" class="${editTool==='rotate'&&!editAim?'active':''}">Rotate · E</button>${light?`<button id="aim-light" class="${editAim?'active':''}">Aim</button>`:''}</div><label class="check-row"><input id="edit-snap" type="checkbox" ${editSnap?'checked':''}> Snap · 10 cm / 15°</label><p class="muted">Select an object or a light handle, then drag a colored axis. Orbit by dragging empty space.</p>`;}
function lightInspector(){const l=currentLight();selectedLight=l.id;return `<span class="eyebrow">LIGHT INSPECTOR</span><h2>${escape(l.name)}</h2><div class="row">${button('light-enable',l.enabled?'On':'Off',l.enabled?'active':'')}${button('light-solo',settings.soloLight===l.id?'Unsolo':'Solo',settings.soloLight===l.id?'active':'')}</div>${slider('rig-intensity','Intensity',l.intensity,0,250,1)}<label class="color-row" for="rig-color">Gel color<input type="color" id="rig-color" value="${l.color}"></label><div class="row"><button data-gel="#ffe0ac">Warm</button><button data-gel="#ffffff">Neutral</button><button data-gel="#8ab8e9">Cool</button></div><h3>Source position</h3>${['X','Y','Z'].map((a,i)=>slider('rig-p'+i,a==='Y'?'Height Y':a,l.position[i],i===1?.2:-12,i===1?8:12,.05,' m')).join('')}${l.type==='spot'?`<h3>Aim at</h3>${['X','Y','Z'].map((a,i)=>slider('rig-t'+i,a,l.target[i],-12,12,.05,' m')).join('')}<div class="row"><button data-aim="0">Ella</button><button data-aim="1">James</button><button data-aim="both">Both</button></div>${slider('rig-angle','Beam spread',l.angle,10,150,1,'°')}${slider('rig-size','Source width',l.size,0,2,.05,' m')}`:'<p class="muted">This practical radiates in every direction.</p>'}<label class="check-row"><input id="rig-shadows" type="checkbox" ${l.shadows?'checked':''}> Cast shadows</label><div class="divider"></div>${button('duplicate-light','Duplicate light','full')}${l.id.startsWith('custom-')?button('delete-light','Delete this light','full'):button('reset-light','Reset this light','full')}<p class="credit">Distance changes falloff. A wider source softens shadow edges using four shadowed samples. Real-time approximation; reflected bounce and diffusion are not simulated.</p>`;}
function renderEditorPanels(){
 if(mode==='lighting'){$('#left').innerHTML=`<span class="eyebrow">THE LIGHTING STAGE</span><h2>Shape what we see.</h2><p>Move every source. Aim the beam. Watch the camera monitor respond.</p><div class="outliner">${settings.lights.map(l=>`<button data-light="${l.id}" class="${l.id===selectedLight?'active':''}"><i style="background:${l.color}"></i><span>${escape(l.name)}<small>${l.type==='point'?'Omnidirectional practical':'Directional source'} · ${l.enabled?'On':'Off'}</small></span></button>`).join('')}</div><div class="row">${button('add-spot','+ Spotlight')}${button('add-point','+ Point')}</div>${toolsPanel(true)}${slider('ambient','Ambient fill',settings.ambient??({blue:.23,tender:.3,uneasy:.12}[settings.mood]),0,.7,.01)}${slider('exposure','Camera exposure',settings.exposure,.3,2.4,.05,' ×')}${button('clear-solo','Show all lights','full')}${button('to-camera','Look through camera','primary full')}`;$('#right').innerHTML=lightInspector();}
 if(mode==='studio'){const items=studio.editor.list();let o=items.find(x=>x.id===selectedObject)||items[0];selectedObject=o.id;$('#left').innerHTML=`<span class="eyebrow">PRODUCTION DESIGN</span><h2>Arrange the scene.</h2><p>Furniture carries its props. Booth seats carry their seated actors.</p><div class="outliner object-list">${items.map(x=>`<button data-object="${x.id}" class="${x.id===o.id?'active':''}"><span>${escape(x.name)}<small>${x.visible?'Visible':'Hidden'}</small></span></button>`).join('')}</div>${button('add-flag','+ Black flag','full')}${toolsPanel()}${button('equipment',settings.equipment?'Hide film equipment':'Show film equipment','full')}${button('export-set','Export 3D set · GLB','full')}`;$('#right').innerHTML=`<span class="eyebrow">OBJECT INSPECTOR</span><h2>${escape(o.name)}</h2><p class="muted">Position is relative to the parent. Cups move with their table or espresso machine.</p><h3>Position</h3>${['X','Y','Z'].map((a,i)=>slider('object-p'+i,a,o.position[i],-12,12,.05,' m')).join('')}<h3>Rotation</h3>${['Pitch X','Turn Y','Roll Z'].map((a,i)=>slider('object-r'+i,a,o.rotation[i],-180,180,1,'°')).join('')}${button('object-visible',o.visible?'Hide object':'Show object','full')}${button('object-reset','Reset placement','full')}${o.custom?button('delete-object','Delete flag','full'):''}<div class="divider"></div><h3>Hero table dressing</h3><div class="row">${['classic','flowers','menu'].map(d=>`<button data-dress="${d}" class="${settings.dress===d?'active':''}">${d[0].toUpperCase()+d.slice(1)}</button>`).join('')}</div><p class="credit">Black flags block direct light and cast shadows. Objects can intersect: placement is manual, without collision or gravity simulation.</p>`;}
}
function changeLight(fn){fn(currentLight());apply();renderPanels();}
function addLight(type,copy=false){if(settings.lights.length>=12){toast('Up to 12 lights per scene. Delete a custom light to add another.');return;}const l=copy?structuredClone(currentLight()):{name:type==='point'?'Point light':'Spotlight',type,position:[2,2.5,2],target:[.5,1,-.6],intensity:30,color:'#ffe0ac',angle:65,size:type==='point'?0:.5,enabled:true,shadows:true};l.id='custom-'+crypto.randomUUID();if(copy){l.name=l.name+' copy';l.position[0]+=.5;}settings.lights.push(l);selectedLight=l.id;apply();studio.editor.select(selectedLight);renderPanels();}
function patchObject(fn){const o=studio.editor.list().find(o=>o.id===selectedObject);const v={position:[...o.position],rotation:[...o.rotation],visible:o.visible};fn(v);settings.objects[o.id]=v;}
function bindEditorUI(){
 $$('[data-light]').forEach(b=>b.onclick=()=>{selectedLight=b.dataset.light;editAim=false;studio.editor.select(selectedLight);renderPanels();});$$('[data-object]').forEach(b=>b.onclick=()=>{selectedObject=b.dataset.object;studio.editor.select(selectedObject);renderPanels();});
 $$('[data-edit]').forEach(b=>b.onclick=()=>{editTool=b.dataset.edit;editAim=false;studio.editor.setTool(editTool);renderPanels();});on('aim-light',()=>{editAim=true;studio.editor.setTool('translate',true);renderPanels();});$('#edit-snap')?.addEventListener('change',e=>{editSnap=e.target.checked;studio.editor.setSnap(editSnap);});
 for(let i=0;i<3;i++){bindSlider('rig-p'+i,v=>currentLight().position[i]=v);bindSlider('rig-t'+i,v=>currentLight().target[i]=v);bindSlider('object-p'+i,v=>patchObject(o=>o.position[i]=v));bindSlider('object-r'+i,v=>patchObject(o=>o.rotation[i]=v));}
 for(const [id,key] of [['rig-intensity','intensity'],['rig-angle','angle'],['rig-size','size']])bindSlider(id,v=>currentLight()[key]=v);
 bindSlider('ambient',v=>settings.ambient=v);$('#rig-color')?.addEventListener('input',e=>{currentLight().color=e.target.value;apply();});$('#rig-shadows')?.addEventListener('change',e=>changeLight(l=>l.shadows=e.target.checked));
 $$('[data-gel]').forEach(b=>b.onclick=()=>changeLight(l=>l.color=b.dataset.gel));$$('[data-aim]').forEach(b=>b.onclick=()=>changeLight(l=>{l.target=b.dataset.aim==='both'?subject(0).map((v,i)=>(v+subject(1)[i])/2):subject(+b.dataset.aim);}));
 on('light-enable',()=>changeLight(l=>l.enabled=!l.enabled));on('light-solo',()=>{settings.soloLight=settings.soloLight===selectedLight?null:selectedLight;apply();renderPanels();});on('clear-solo',()=>{settings.soloLight=null;apply();renderPanels();});
 on('add-spot',()=>addLight('spot'));on('add-point',()=>addLight('point'));on('duplicate-light',()=>addLight(currentLight().type,true));on('delete-light',()=>{settings.lights=settings.lights.filter(l=>l.id!==selectedLight);if(settings.soloLight===selectedLight)settings.soloLight=null;selectedLight='key';apply();studio.editor.select(selectedLight);renderPanels();});on('reset-light',()=>{const l=normalizeScene(defaults()).lights.find(l=>l.id===selectedLight);if(l){Object.assign(currentLight(),l);apply();renderPanels();}});
 on('add-flag',()=>{if(settings.flags.length>=12){toast('Up to 12 flags per scene.');return;}selectedObject='flag-'+crypto.randomUUID();settings.flags.push({id:selectedObject});apply();studio.editor.select(selectedObject);renderPanels();});on('delete-object',()=>{settings.flags=settings.flags.filter(f=>f.id!==selectedObject);delete settings.objects[selectedObject];selectedObject='table-1';apply();studio.editor.select(selectedObject);renderPanels();});on('object-visible',()=>{patchObject(o=>o.visible=!o.visible);apply();renderPanels();});on('object-reset',()=>{delete settings.objects[selectedObject];apply();renderPanels();});
 on('edit-rig',()=>setMode('lighting'));
}
